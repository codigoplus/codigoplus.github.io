Cómo implementarlo real en Netlify (anti-spam por IP)

Crea /netlify/functions/dejar-plus.js:

// netlify/functions/dejar-plus.js
import { getStore } from "@netlify/blobs";
import crypto from "crypto";

export default async (req) => {
  if (req.method !== "POST") return new Response("Method Not Allowed", {status:405});
  
  const ip = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() || "unknown";
  const hashIp = crypto.createHash("sha256").update(ip).digest("hex");
  
  const store = getStore("codigos-plus");
  const existe = await store.get(hashIp);
  if (existe) {
    return new Response(JSON.stringify({error: "Ya dejaste tu +1 desde esta IP"}), {status:409});
  }

  const { nombre, plus } = await req.json();
  if(!nombre || !plus) return new Response("Faltan datos", {status:400});

  await store.setJSON(hashIp, { nombre, plus, fecha: new Date().toISOString(), ipMask: ip.slice(0,3)+"•••" });
  return new Response(JSON.stringify({ok:true}), {status:200});
}
